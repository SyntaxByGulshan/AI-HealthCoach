# 2) Imports
import os
import numpy as np
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.pipeline import Pipeline
from sklearn.compose import ColumnTransformer
from sklearn.preprocessing import OneHotEncoder, StandardScaler
from sklearn.ensemble import RandomForestRegressor
from sklearn.metrics import mean_absolute_error, mean_squared_error, r2_score
import joblib
import pickle
import math
import tensorflow as tf
from tensorflow.keras import layers, models, callbacks, optimizers

# Reproducibility
RANDOM_STATE = 42
np.random.seed(RANDOM_STATE)
tf.random.set_seed(RANDOM_STATE)

# 3) Configuration - change these as needed
CSV_PATH = "/content/calorie_prediction_dataset_50k.csv"   # <-- put your CSV here
TARGET_COL = "calories"
SKLEARN_MODEL_PKL = "/content/calorie_model.pkl"
SKLEARN_MODEL_JOBLIB = "/content/calorie_model.joblib"
KERAS_MODEL_PATH = "/content/keras_calorie_model.h5"
DO_TRAIN_KERAS = True          # set False to skip Keras NN training
KERAS_EPOCHS = 50              # if >30, checkpointing will be used per your request
KERAS_BATCH_SIZE = 32

# 4) Load data
if not os.path.exists(CSV_PATH):
    raise FileNotFoundError(f"CSV not found at {CSV_PATH}. Upload your CSV file to Colab and name it 'data.csv' or change CSV_PATH")

df = pd.read_csv(CSV_PATH)
print("Dataset loaded. Shape:", df.shape)
print("Columns:", df.columns.tolist())

# 5) Minimal sanity checks & preview
if TARGET_COL not in df.columns:
    raise ValueError(f"Target column '{TARGET_COL}' not found in CSV. Make sure your dataset has a '{TARGET_COL}' column.")
display(df.head())

# 6) Feature engineering:
# We will use: age, gender, weight, height, activity_level, goal (categorical)
# If user supplied 'goal_weight' column, we'll add a numeric feature 'weight_diff' = goal_weight - weight.
# Also compute BMI to help model generalize (weight in kg, height in cm assumed in your sample).
df = df.copy()

# Check required columns presence; raise helpful errors if missing.
required_cols = ["age", "gender", "weight", "height"]
missing = [c for c in required_cols if c not in df.columns]
if missing:
    raise ValueError(f"Missing required columns: {missing}. Your dataset must contain: {required_cols}")

# Make BMI (weight in kg, height in cm)
# Avoid divide-by-zero or missing heights.
df["height_m"] = df["height"].replace(0, np.nan) / 100.0
df["BMI"] = df["weight"] / (df["height_m"] ** 2)
df["BMI"] = df["BMI"].replace([np.inf, -np.inf], np.nan)

# Optional: goal_weight -> weight_diff
if "goal_weight" in df.columns:
    df["weight_diff"] = df["goal_weight"] - df["weight"]
else:
    # if not present, create weight_diff = 0 so pipeline shape is consistent if used elsewhere
    df["weight_diff"] = 0.0

# Fill / handle missing values for BMI and height_m
df["BMI"] = df["BMI"].fillna(df["BMI"].median())
df["height_m"] = df["height_m"].fillna(df["height_m"].median())

# For categorical columns, ensure they exist (activity_level and goal are present in sample but may be missing)
for cat in ["activity_level", "goal"]:
    if cat not in df.columns:
        # create a default category if missing, but warn
        print(f"Warning: column '{cat}' not found. Creating default '{cat}' = 'unknown'.")
        df[cat] = "unknown"

# 7) Select features and target
feature_cols = ["age", "gender", "weight", "height", "BMI", "weight_diff", "activity_level", "goal"]
X = df[feature_cols]
y = df[TARGET_COL].astype(float)

# 8) Train/test split
X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.20, random_state=RANDOM_STATE)
print("Train/Test sizes:", X_train.shape, X_test.shape)

# 9) Preprocessing pipeline
numeric_features = ["age", "weight", "height", "BMI", "weight_diff"]
categorical_features = ["gender", "activity_level", "goal"]

numeric_transformer = Pipeline([
    ("scaler", StandardScaler())
])

categorical_transformer = Pipeline([
    ("ohe", OneHotEncoder(handle_unknown="ignore", sparse_output=False)) # Changed sparse to sparse_output
])

preprocessor = ColumnTransformer(
    transformers=[
        ("num", numeric_transformer, numeric_features),
        ("cat", categorical_transformer, categorical_features),
    ],
    remainder="drop",  # drop any other columns
    sparse_threshold=0
)

# Combine with a RandomForest regressor
sklearn_pipeline = Pipeline([
    ("preprocessor", preprocessor),
    ("regressor", RandomForestRegressor(n_estimators=200, random_state=RANDOM_STATE, n_jobs=-1))
])

# 10) Train sklearn model
print("Training sklearn RandomForest model (this may take a minute)...")
sklearn_pipeline.fit(X_train, y_train)

# 11) Evaluate sklearn model
y_pred = sklearn_pipeline.predict(X_test)
mae = mean_absolute_error(y_test, y_pred)
rmse = math.sqrt(mean_squared_error(y_test, y_pred))
r2 = r2_score(y_test, y_pred)

print(f"Sklearn RandomForest results on test set -> MAE: {mae:.2f}, RMSE: {rmse:.2f}, R2: {r2:.4f}")

# 12) Save sklearn pipeline to .pkl and .joblib
print(f"Saving sklearn pipeline to {SKLEARN_MODEL_PKL} and {SKLEARN_MODEL_JOBLIB} ...")
with open(SKLEARN_MODEL_PKL, "wb") as f:
    pickle.dump(sklearn_pipeline, f)
joblib.dump(sklearn_pipeline, SKLEARN_MODEL_JOBLIB)
print("Saved sklearn model files.")

# 13) Optional: train a Keras neural network (with checkpoint if epochs > 30)
if DO_TRAIN_KERAS:
    print("Preparing and training Keras NN (this is optional and may be slower).")
    # Build a small pipeline to produce numeric numpy arrays for NN
    preprocessor_for_nn = preprocessor  # same preprocessor can be used to transform to numpy
    X_train_trans = preprocessor_for_nn.fit_transform(X_train)
    X_test_trans = preprocessor_for_nn.transform(X_test)
    input_dim = X_train_trans.shape[1]
    print("Input dim for NN:", input_dim)

    # Build model
    def build_keras_model(input_dim):
        model = models.Sequential([
            layers.Input(shape=(input_dim,)),
            layers.Dense(128, activation="relu"),
            layers.Dense(64, activation="relu"),
            layers.Dense(32, activation="relu"),
            layers.Dense(1, activation="linear")
        ])
        model.compile(optimizer=optimizers.Adam(learning_rate=1e-3),
                      loss="mse",
                      metrics=["mae"])
        return model

    keras_model = build_keras_model(input_dim)

    # Callbacks: EarlyStopping + ModelCheckpoint when epochs > 30
    cb = []
    cb.append(callbacks.EarlyStopping(monitor="val_loss", patience=10, restore_best_weights=True))
    if KERAS_EPOCHS > 30:
        # Save best weights to file during training
        checkpoint_cb = callbacks.ModelCheckpoint(
            filepath=KERAS_MODEL_PATH,
            monitor="val_loss",
            save_best_only=True,
            save_weights_only=False
        )
        cb.append(checkpoint_cb)
        print(f"KERAS_EPOCHS={KERAS_EPOCHS} > 30: ModelCheckpoint enabled, saving best model to {KERAS_MODEL_PATH}")

    # Fit
    history = keras_model.fit(
        X_train_trans, y_train.values,
        validation_data=(X_test_trans, y_test.values),
        epochs=KERAS_EPOCHS,
        batch_size=KERAS_BATCH_SIZE,
        callbacks=cb,
        verbose=2
    )

    # If checkpoint wasn't used (epochs <= 30), still save final model
    if not os.path.exists(KERAS_MODEL_PATH):
        keras_model.save(KERAS_MODEL_PATH)
        print(f"Keras model saved to {KERAS_MODEL_PATH}")

    # Evaluate
    nn_preds = keras_model.predict(X_test_trans).ravel()
    mae_nn = mean_absolute_error(y_test, nn_preds)
    rmse_nn = math.sqrt(mean_squared_error(y_test, nn_preds))
    r2_nn = r2_score(y_test, nn_preds)
    print(f"Keras NN results on test set -> MAE: {mae_nn:.2f}, RMSE: {rmse_nn:.2f}, R2: {r2_nn:.4f}")

    print(f"Keras model stored at: {KERAS_MODEL_PATH}")

# 14) Example inference function (use the sklearn pipeline for quick inference)
def predict_calories_from_input(pipeline, age, gender, weight, height, activity_level="moderate", goal="maintain", goal_weight=None):
    """
    Returns predicted calories from provided inputs using the saved sklearn pipeline.
    If goal_weight provided, weight_diff is computed; else weight_diff=0.
    """
    height_m = (height / 100.0) if height is not None else None
    BMI = weight / (height_m ** 2) if (height_m and height_m > 0) else 0.0
    weight_diff = 0.0 if goal_weight is None else (goal_weight - weight)
    row = pd.DataFrame([{
        "age": age,
        "gender": gender,
        "weight": weight,
        "height": height,
        "BMI": BMI,
        "weight_diff": weight_diff,
        "activity_level": activity_level,
        "goal": goal
    }])
    return pipeline.predict(row)[0]

# Quick demo using the first row of test set
sample = X_test.iloc[0]
pred_example = predict_calories_from_input(sklearn_pipeline,
                                           age=float(sample["age"]),
                                           gender=sample["gender"],
                                           weight=float(sample["weight"]),
                                           height=float(sample["height"]),
                                           activity_level=sample["activity_level"],
                                           goal=sample["goal"],
                                           goal_weight=None)
print("Example predicted calories (sklearn pipeline) for one test sample:", round(pred_example, 2))
